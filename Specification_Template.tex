% DMN's own version of the Specification_Template.tex of mdsa-omg-core (a local
% file replaces the core one). It keeps DMN's structure, so that the PDF is
% equivalent to the DMN document made with Word:
% - the cover page, copyrights and notices of DMN (\maketitle and \copyrights in
%   _Specification_AuthorSettings.tex, notices in 00_Notices.md), the table of
%   contents, and the preface of DMN (00_Preface.md); no lists of figures and
%   tables
% - page breaks and blank pages as in DMN (see \dmnblankpage below)
% - Clauses 1 to 4 as in DMN: 01_Scope.md, 02_Conformance.md, the references
%   (Clause 3, from specification.bib) and 04_Additional_information.md, then
%   the clauses listed in 07+_Technical_Content.tex and the annexes. There are
%   no clauses for terms and symbols, and no submission clause.
\usepackage{omg_specification}

%% User added packages
\usepackage{import}

\InputIfFileExists{pandoc-preamble.tex}{}{}

\buildauthornames

% Each clause from Clause 5 on, the page "ANNEXES" and the annexes start on a
% new page. With \dmnwordpagestrue (for a page-by-page comparison with the Word
% PDF of DMN 1.7 Beta 1, together with the page starts of specification.lua,
% see Makefile.mdsa), they start on the page they start on in the Word PDF,
% preceded by a page "This page intentionally left blank.": if the clause
% before ends later, the blank page is left out; if it ends earlier, empty
% pages fill the gap. (\dmnblankpages is switched on before Clause 5.)
\newif\ifdmnwordpages
\newif\ifdmnblankpages
\newcounter{dmnchapter}
\newcommand{\dmnstartpage}{%
  \ifcase\value{dmnchapter}\or
  10\or 22\or 53\or 67\or 85\or 91\or 173\or 180\or 224\or 227\or % Clauses 5 to 14
  241\or 243\or 248\else 0\fi}                                     % ANNEXES, A, B
\newcommand{\dmnblankpage}{%
  \clearpage
  \ifdmnwordpages\dmnwordpage\fi}
\newcommand{\dmnwordpage}{%
  \stepcounter{dmnchapter}%
  \edef\dmntarget{\dmnstartpage}%
  \loop\ifnum\numexpr\value{page}+1<\dmntarget\relax
    \null\clearpage
  \repeat
  \ifnum\numexpr\value{page}+1=\dmntarget\relax
    \ifnum\value{dmnchapter}<13
      \vspace*{125pt}{\centering This page intentionally left blank.\par}%
    \else\null\fi
    \clearpage
  \fi}
% Clauses do not start on a new page by themselves (Clause 2 follows Clause 1 on
% its page, as in DMN); the page breaks are below and in \dmnblankpage
\makeatletter
\patchcmd{\chapter}{\if@openright\cleardoublepage\else\clearpage\fi}{}{}{}
\makeatother
\pretocmd{\chapter}{\par\ifdmnblankpages\dmnblankpage\fi}{}{}
